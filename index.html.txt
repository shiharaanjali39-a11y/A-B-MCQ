<script>
const answers = {1: 2, 11: 5, 21: 4, 31: 1, 41: 2, 2: 3, 12: 1, 22: 5, 32: 3, 42: 2, 3: 1, 13: 5, 23: 2, 33: 2, 43: 4, 4: 5, 14: 5, 24: 3, 34: 3, 44: 1, 5: 3, 15: 5, 25: 3, 35: 1, 45: 4, 6: 5, 16: 1, 26: 4, 36: 4, 46: 4, 7: 5, 17: 1, 27: 2, 37: 5, 47: 2, 8: 4, 18: 4, 28: 3, 38: 1, 48: 2, 9: 1, 19: 3, 29: 5, 39: 4, 49: 1, 10: 3, 20: 4, 30: 2, 40: 3, 50: 3};
const quizDiv = document.getElementById('quiz');
for(let i=1;i<=50;i++){
  let card = document.createElement('div');
  card.className='card';
  card.innerHTML = `<div class="q-title">ප්‍රශ්නය ${i} - Question ${i}</div><div class="options">${[1,2,3,4,5].map(opt=>`<label id="label-${i}-${opt}"><input type="radio" name="q${i}" value="${opt}"> (${opt})</label>`).join('')}</div><div class="result" id="result-${i}"></div>`;
  quizDiv.appendChild(card);
}
function checkAll(){
  let score=0, attempted=0;
  for(let i=1;i<=50;i++){
    let sel = document.querySelector(`input[name="q${i}"]:checked`);
    let res = document.getElementById(`result-${i}`);
    for(let o=1;o<=5;o++) document.getElementById(`label-${i}-${o}`).classList.remove('correct','wrong');
    if(!sel){ res.innerHTML='⚠️ තෝරලා නෑ'; res.style.color='#856404'; continue; }
    attempted++;
    let correct = answers[i];
    let val = parseInt(sel.value);
    if(val===correct){
      res.innerHTML=`✔ හරි! උත්තරය (${correct})`; res.style.color='green';
      document.getElementById(`label-${i}-${val}`).classList.add('correct'); score++;
    } else {
      res.innerHTML=`✖ වැරදි! හරි උත්තරය (${correct})`; res.style.color='red';
      document.getElementById(`label-${i}-${val}`).classList.add('wrong');
      document.getElementById(`label-${i}-${correct}`).classList.add('correct');
    }
  }
  let box = document.getElementById('scoreBox');
  box.style.display='block';
  box.innerHTML=`ඔයාගේ ලකුණු: ${score} / 50<br>Attempted: ${attempted}`;
}
function resetAll(){
  document.querySelectorAll('input[type=radio]').forEach(r=>r.checked=false);
  document.querySelectorAll('.result').forEach(r=>r.innerHTML='');
  document.querySelectorAll('label').forEach(l=>l.classList.remove('correct','wrong'));
  document.getElementById('scoreBox').style.display='none';
}
